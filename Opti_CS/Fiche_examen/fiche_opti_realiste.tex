% Fiche d'examen -- Optimisation : VERSION "VRAIMENT MANUSCRITE".
% Meme contenu (fiche_contenu.tex) et meme habillage que fiche_opti_manuscrite.tex (Patrick Hand,
% formules en Euler), avec en plus les irregularites d'une vraie ecriture. Deux etapes :
%  1) ce fichier (LuaLaTeX) + ../../outils_manuscrit/manuscrit.lua : espacement aleatoire entre
%     lettres et mots, pas de cesure, chaque glyphe (texte et maths) legerement tourne / etire /
%     incline / decale ; soulignes de titres, triangles "!", etoiles, fleches et doubles barres
%     (R, Z) traces a main levee ; trois ratures. Les encres sont des couleurs "marqueurs"
%     (noir = texte, bleu pur = titres, rouge pur, vert pur, cyan = violet, magenta = crayon)
%     que l'etape 2 remplace par de vraies couleurs de stylo.
%  2) ../../outils_manuscrit/reencrage.py : rendu 600 dpi, deformations elastiques multi-echelles,
%     reencrage au stylo (largeur et pression variables), traits de colonnes au crayon.
% Compilation (depuis ce dossier) :
%   lualatex fiche_opti_realiste.tex
%   python3 ../../outils_manuscrit/reencrage.py fiche_opti_realiste.pdf

% (pas de "rand" dans un rayon polaire : TikZ y lit "and")
\newcommand{\handstar}{\tikz[main,baseline={(0,-0.3em)}]{\pgfmathsetmacro{\hr}{90+9*rand}%
  \draw (\hr:{0.33+0.04*(2*rnd-1)}) -- (\hr+144:{0.33+0.04*(2*rnd-1)}) -- (\hr+288:{0.33+0.04*(2*rnd-1)})
     -- (\hr+432:{0.33+0.04*(2*rnd-1)}) -- (\hr+576:{0.33+0.04*(2*rnd-1)}) -- (\hr+722+7*rand:{0.35+0.05*(2*rnd-1)});}}
\newcommand{\handarrow}{\tikz[main,baseline=0pt]{\draw (0,{0.28+0.03*rand}) .. controls (0.36,{0.30+0.04*rand}) .. (0.72,0.29);
     \draw ({0.52+0.03*rand},{0.47+0.03*rand}) -- (0.73,0.29) -- ({0.53+0.03*rand},{0.12+0.03*rand});}}
% triangle "attention" avec un point d'exclamation, trace a la main
\newcommand{\handwarn}{\tikz[main,baseline=-0.04em]{%
  \draw ({0.02*rand},{0.02*rand}) -- ({0.45+0.03*rand},{0.86+0.03*rand}) -- ({0.90+0.03*rand},{0.02*rand}) -- ({-0.06+0.03*rand},{0.03*rand});
  \draw ({0.45+0.02*rand},0.62) -- ({0.45+0.02*rand},{0.30+0.03*rand}); \fill ({0.45+0.02*rand},0.15) circle (0.045);}}
% lettre a double barre (R, Z, N...) : la lettre manuscrite + un second trait a la main
\newdimen\bbw \newdimen\bbh
\newcommand{\HandBBbox}[1]{\setbox0\hbox{#1}\bbw=\wd0 \bbh=\fontcharht\font`#1\relax
  \rlap{\box0}%
  \makebox[0pt][l]{\tikz[baseline=0pt,line width=0.48pt,line cap=round]{%
    \ifx Z#1\draw ({0.12*\bbw+0.15em},{0.05*\bbh}) .. controls ({0.5*\bbw+0.15em+0.02em*rand},{0.5*\bbh})
        .. ({0.86*\bbw+0.08em},{0.95*\bbh});%
    \else\draw ({0.21em+0.012em*rand},{0.04*\bbh}) .. controls ({0.21em+0.06*\bbh+0.02em*rand},{0.5*\bbh})
        .. ({0.21em+0.12*\bbh+0.02em*rand},{0.97*\bbh});\fi}}%
  \kern\bbw}
\renewcommand{\mathbb}[1]{\text{\HandBBbox{#1}}}
% soulignement de titre : un trait a main levee (depart avant le titre, longueur et pente variables)
\newcommand{\handunder}[1]{\draw[cT,line width=0.55pt,line cap=round]
  ({-0.8pt-0.8pt*rnd},{-1.7pt+0.25pt*rand}) .. controls ({0.5*#1},{-1.6pt+0.6pt*rand}) .. ({#1},{-1.7pt+0.5pt*rand});}
% rature : le texte barre de deux traits rapides, puis on continue
\newcommand{\raturebox}[1]{\tikz[baseline=(r.base)]{\node[inner sep=0pt,outer sep=0pt] (r) {#1};
  \draw[line width=0.5pt,line cap=round] ($(r.west)+({-0.6pt},{-0.5pt+0.3pt*rand})$) -- ($(r.east)+({0.6pt},{0.5pt+0.3pt*rand})$);
  \draw[line width=0.5pt,line cap=round] ($(r.west)+({-0.4pt},{0.6pt+0.3pt*rand})$) -- ($(r.east)+({0.7pt},{-0.3pt+0.3pt*rand})$);}\kern0.18em}
