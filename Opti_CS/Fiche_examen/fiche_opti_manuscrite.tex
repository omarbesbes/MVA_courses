\Sec{0. Vocabulaire}
\B $u$ \textbf{min local} : $u\in K$, $\exists\delta>0$, $\nr{v-u}<\delta,v\in K\Rightarrow J(v)\ge J(u)$. \textbf{global} : $J(v)\ge J(u)\ \forall v\in K$.
\B $\inf_K J$ = plus grand minorant ($-\infty$ si non minorée, $+\infty$ si $K=\emptyset$). \textbf{Suite minimisante} $u_n\in K$, $J(u_n)\to\inf_K J$ : existe toujours.
\B \textbf{Coercive} (« infinie à l'infini ») : $\nr{u_n}\to\infty\Rightarrow J(u_n)\to\infty$ (gratuit si $K$ borné). Variante : $\inf_K J<\liminf_{R\to\infty}\inf_{\nr v\ge R}J$.
\B \textbf{sci} : $u_n\to u\Rightarrow \liminf J(u_n)\ge J(u)$.

\Sec{1. Existence -- unicité}
\B \St\Th{Th 2.2.1} (dim finie) $K\subset\R^N$ \textbf{fermé} $\neq\emptyset$, $J$ \textbf{continue (ou sci) + coercive} $\Rightarrow\exists$ min ; toute suite minimisante a une sous-suite $\to$ un min. \emph{Preuve} : $(u_n)$ bornée (coerc.) $\to$ Bolzano--W. $\to u\in K$ (fermé) $\to J(u)\le\liminf J(u_{n_k})=\inf$.
\B Contre-ex. : $K$ non fermé ($\inf_{]0,1]}x$), $J=e^x$ sur $\R$ (minorée, pas de min). Dim. $\infty$ : fermé borné $\not\Rightarrow$ compact ; $\ell^2$, $J(x)=(\nr x^2-1)^2+\sum x_i^2/i$, $x^n=e_n$ : $J=1/n\to0$ non atteint.
\B \St\Th{Prop 2.3.4} $J$ convexe sur $K$ convexe : \textbf{min local = global}, ens. des min convexe ; $J$ \textbf{strict.} convexe $\Rightarrow$ \textbf{au plus un} min.
\B \Th{Th 2.3.9} (Hilbert) $K$ convexe fermé $\ne\emptyset$, $J$ convexe, continue (sci), coercive $\Rightarrow\exists$ min (conv. faible ; convexe fermé $\Rightarrow$ faiblement fermé). Pas d'unicité.
\B \St\Th{Th 2.3.8} $K$ convexe fermé $\ne\emptyset$, $J$ \textbf{$\alpha$-convexe continue} $\Rightarrow\exists!$ min $u$ et $\nr{v-u}^2\le\frac4\alpha(J(v)-J(u))\ \forall v\in K$ ; toute suite minimisante $\to u$. (Pas besoin de coercivité : $J\ge\gamma\nr v^2+\eta$.)
\B \textbf{Convexe} : $J(\theta u+(1-\theta)v)\le\theta J(u)+(1-\theta)J(v)$. \textbf{Strict.} : $<$ si $u\ne v$, $\theta\in]0,1[$. \textbf{$\alpha$-convexe} : $\ldots-\frac{\alpha\theta(1-\theta)}2\nr{u-v}^2$ $\Leftrightarrow J-\frac\alpha2\nr\cdot^2$ convexe $\Leftrightarrow$ ($\theta=\frac12$) $J(\frac{u+v}2)\le\frac{J(u)+J(v)}2-\frac\alpha8\nr{u-v}^2$. $\alpha$-conv $\Rightarrow$ strict $\Rightarrow$ convexe.
\B Stables : $J_1+J_2$, $\max(J_1,J_2)$, $\lambda J$ ($\lambda>0$), $\varphi\circ J$ ($\varphi$ convexe \textbf{croissante}), $\sup_i$ affines. Convexe $\Leftrightarrow$ épigraphe convexe. Convexe (majorée loc.) $\Rightarrow$ continue ; $\ge$ une minorante affine. \emph{Exemples} : normes, $\nr{Ax-b}^2$, $e^{x}$, $-\log x$, $x\log x$, $|x|^p$ ($p\ge1$), $\log\sum e^{x_i}$. \textbf{Ensembles} : $\{F\le0\}$ convexe si $F$ convexe ; $\{F=0\}$ convexe si $F$ affine ; intersection, image affine, boule, demi-espace, polyèdre.
\B \St Différentiable : convexe $\Leftrightarrow J(v)\ge J(u)+\sca{J'(u)}{v-u}$ $\Leftrightarrow\sca{J'(u)-J'(v)}{u-v}\ge0\Leftrightarrow J''(u)(w,w)\ge0$.\\
$\alpha$-conv $\Leftrightarrow J(v)\ge J(u)+\sca{J'(u)}{v-u}+\frac\alpha2\nr{v-u}^2\Leftrightarrow\sca{J'(u)-J'(v)}{u-v}\ge\alpha\nr{u-v}^2\Leftrightarrow J''\ge\alpha\,\mathrm{Id}$.
\B \St $J'$ \textbf{$L$-Lipschitz} $\Rightarrow J(v)\le J(u)+\sca{J'(u)}{v-u}+\frac L2\nr{v-u}^2$ ($C^2$ : $J''\le L\,\mathrm{Id}$). Donc $\alpha\mathrm{Id}\le J''\le L\mathrm{Id}$ : $J$ coincée entre 2 paraboles tangentes ; $\kappa=L/\alpha\ge1$. Co-coercivité (convexe) : $\nr{J'(v)-J'(w)}^2\le L\sca{J'(v)-J'(w)}{v-w}$.
\B \St\textbf{Quadratique} $J=\frac12Ax\cdot x-b\cdot x$ ($A$ sym.) : $J'=Ax-b$, $J''=A$ ; convexe $\Leftrightarrow A\ge0$ ; strict $\Leftrightarrow A>0\Leftrightarrow\alpha$-conv., $\alpha=\lambda_{\min}$, $L=\lambda_{\max}$ ; min $x=A^{-1}b$. Hilbert : $J=\frac12a(u,u)-\ell(u)$, $a$ coercive $\Rightarrow\alpha$-convexe.

\Sec{2. Dérivées}
\B \St\textbf{Fréchet} : $J(u+w)=J(u)+\ell(w)+o(\nr w)$, $\ell\in V'$ ; Riesz : $\ell(w)=\sca{p}{w}$, $p=J'(u)=\nabla J$. \textbf{Gâteaux} : $\lim_{\delta\to0^+}\frac{J(u+\delta w)-J(u)}\delta=\ell(w)$ (Fréchet $\Rightarrow$ Gâteaux, pas $\Leftarrow$).
\B Taylor : $J(u+w)=J(u)+\sca{J'(u)}w+\frac12J''(u)(w,w)+o(\nr w^2)$.
\Mt Calcul : développer $J(u+\eps w)$, garder le terme linéaire en $w$. $J=\frac12a(u,u)-\ell(u)$ : $\sca{J'(u)}w=a(u,w)-\ell(w)$, $J''(v,w)=a(v,w)$.
\B $\nabla(x\cdot Ax)=(A+A\T)x$ ; $\nabla\nr{Ax-b}^2=2A\T(Ax-b)$, Hess $2A\T A$ ; $\nabla\nr x=x/\nr x$ ; $\nabla(J\circ F)=F'(u)\T J'(F(u))$ ; $\int f\log f\to\log f+1$ ; $\int\frac\mu2|u'|^2-fu$ ($u(0)=u(L)=0$) $\to J'(u)w=\int\mu u'w'-fw$ $\Rightarrow-\mu u''=f$.
\B Moindres carrés $\min\nr{Ax-b}^2$ : existe toujours ; $A\T Ax=A\T b$ ; unique $\Leftrightarrow\ker A=\{0\}$.

\Sec{3. Optimalité -- cas convexe / sans contrainte}
\B \St $K=V$ ou $u$ intérieur : \textbf{Euler} $J'(u)=0$. Ordre 2 : nécessaire $J''(u)\ge0$ ; suffisant : $J'(u)=0$ et $J''(v)\ge0$ $\forall v$ voisin (dim finie : $J''(u)>0\Rightarrow$ min local strict).
\B \St\Th{Th 2.5.1} $K$ \textbf{convexe} : min local $\Rightarrow$ \textbf{inéquation d'Euler} $\sca{J'(u)}{v-u}\ge0\ \forall v\in K$ ; \textbf{CNS si $J$ convexe}. $\Leftrightarrow u=P_K(u-\mu J'(u))\ \forall\mu>0$.
\B \textbf{Projection} (Th 5.1.3) $K$ conv. fermé : $x_K\in K$, $\sca{x_K-x}{x_K-y}\le0\ \forall y\in K$ ; $P_K$ 1-Lipschitz. Formules : $\{Bx=c\}$ : $x+B\T(BB\T)^{-1}(c-Bx)$ ; pavé $\prod[a_i,b_i]$ : $y_i=\min(\max(a_i,x_i),b_i)$ ; $\R^M_+$ : $x^+$ ; boule $B(0,R)$ : $x\min(1,R/\nr x)$ ; $\{a\cdot x=\beta\}$ : $x-\frac{a\cdot x-\beta}{\nr a^2}a$ ; $\{\sum x_i=1\}$ : $x-\frac{\sum x_i-1}N\mathbf1$.
\B Cône tangent $K(u)=\{w:\exists u_n\in K\to u,\eps_n\searrow0,\frac{u_n-u}{\eps_n}\to w\}$ ; CN générale : $\sca{J'(u)}w\ge0\ \forall w\in K(u)$ ($K(u)=V$ si $u$ intérieur).
\B $J_1$ dérivable $+J_2$ convexe : $u$ min $\Leftrightarrow\sca{J_1'(u)}{v-u}+J_2(v)-J_2(u)\ge0$.

\Sec{4. Égalités : Lagrange}
\B \St\Th{Th 2.5.6} $K=\{F_i(v)=0,\,1\le i\le M\}$, $F_i$ $C^1$ près de $u$, \textbf{$(F_i'(u))$ libres} (cas régulier). $u$ min local $\Rightarrow\exists!\lambda\in\R^M$ : $J'(u)+\sum\lambda_iF_i'(u)=0$. Ici $K(u)=\bigcap\ker F_i'(u)$.
\B \pg{Non régulier} : $J(v)=v$, $F(v)=v^2$ : $K=\{0\}$, $F'(0)=0$, $J'(0)=1\ne0$ : pas de $\lambda$ !
\B $\Lg(v,\mu)=J+\mu\cdot F$ : $\partial_v\Lg=0$, $\partial_\mu\Lg=F=0$ ($N+M$ éq.). $\inf_{F=0}J=\inf_v\sup_{\mu\in\R^M}\Lg$.
\B \St\Th{Prop 2.5.13} (ordre 2, $C^2$) $\big(J''(u)+\sum\lambda_iF_i''(u)\big)(w,w)\ge0\ \forall w\in\ker F'(u)$. \pg{Hessienne du LAGRANGIEN, sur le tangent seulement.} $>0$ ($w\ne0$) $\Rightarrow$ min local strict.
\B Sensibilité : $F=\eps$ : $\lambda_i=-\partial J_{\min}/\partial\eps_i$ (prix marginal).
\B $\min\frac12Ax\cdot x-b\cdot x\st Bx=c$ ($A>0$, rg $B=M$) : $Ax-b+B\T\lambda=0$, $x=A^{-1}(b-B\T\lambda)$, $(BA^{-1}B\T)\lambda=BA^{-1}b-c$.
\B $\min_{\nr x=1}Ax\cdot x$ : $Ax=-\lambda x$ $\Rightarrow$ vecteur propre de $\lambda_{\min}$. $\sup_{Ax\cdot x\le1}b\cdot x$ : $x=\frac{A^{-1}b}{\sqrt{A^{-1}b\cdot b}}$, val. $\sqrt{A^{-1}b\cdot b}$. Entropie $\min\sum p_i\log p_i$, $\sum p_i=1$ : $p_i=\frac1n$.

\Sec{5. Inégalités : KKT}
\B $K=\{F_i(v)\le0\}$ ; \textbf{actives} $I(u)=\{i:F_i(u)=0\}$.
\B \St\textbf{Qualifiées} en $u$ (Déf 2.5.15) : $\exists w$, $\forall i\in I(u)$ : $\sca{F_i'(u)}w<0$, ou $=0$ si $F_i$ affine. \textbf{Suffit} : (a) $(F_i'(u))_{i\in I(u)}$ libre ; (b) toutes $F_i$ affines ($w=0$) ; (c) \textbf{Slater} : $F_i$ convexes et $\exists v$ : $F_i(v)<0$ (ou $=0$ et $F_i$ affine) -- sans connaître $u$.
\B \St\Th{Th 2.5.17 (KKT)} $J,F_i$ dérivables en $u$, qualif. en $u$, $u$ min local $\Rightarrow\exists\lambda\in\R^M$ :
\[J'(u)+\textstyle\sum\lambda_iF_i'(u)=0,\ \lambda\ge0,\ F(u)\le0,\ \lambda\cdot F(u)=0\]
(écarts compl. : $F_i(u)<0\Rightarrow\lambda_i=0$). Farkas : $\sca pw\ge0\ \forall w\in\{\sca{a_i}w\le0\}\Leftrightarrow p=-\sum\lambda_ia_i$, $\lambda\ge0$.
\B \Th{Th 2.5.24 (mixte)} $G_j(v)=0$, $F_i(v)\le0$ ; qualif : $(G_j'(u))$ libre et $\exists w\in\bigcap\ker G_j'(u)$, $\sca{F_i'(u)}w<0\ \forall i\in I(u)$ (ou $(G_j',F_{i\in I}')$ libre). Alors $J'+\sum\mu_jG_j'+\sum\lambda_iF_i'=0$, $\mu\in\R^N$ \textbf{libre}, $\lambda\ge0$, $\lambda\cdot F=0$.
\Pg \textbf{Non qualifié} $\Rightarrow$ KKT peut échouer : $\min-x_1\st x_2\le(1-x_1)^3$, $x_2\ge0$ : min en $(1,0)$, gradients actifs $(0,1),(0,-1)$ liés, $J'=(-1,0)$ : aucun $\lambda$.
\Pg Signe : \textbf{min} et contraintes \textbf{$F\le0$} $\Rightarrow+\lambda F'$, $\lambda\ge0$. $\max J=-\min(-J)$ ; $F\ge0\to-F\le0$ ; $x\ge0\to-x\le0$ : $J'(x)-\lambda=0$.
\Mt\textbf{Résoudre à la main} : (1) existence (compact / coercive), (2) qualification, (3) système KKT, (4) \textbf{cas} sur l'ensemble actif ($2^M$), vérifier $\lambda\ge0$ et $F\le0$, (5) comparer les $J$ ou convexité $\Rightarrow$ suffisant.
\B $\min\frac12Ax\cdot x-b\cdot x\st Bx\le c$ : $Ax-b+B\T p=0$, $p\ge0$, $p\cdot(Bx-c)=0$. Markowitz $\min\frac12x\T Vx$, $\sum x_i=1$, $x\ge0$ : $Vx+\mu\mathbf1-\lambda=0$, $\lambda\ge0$, $\lambda_ix_i=0$.
\B Sensibilité ($F\le\eps$) : $\lambda_i=-\partial J_{\min}/\partial\eps_i\ge0$. Var. d'écart : $F_i+z_i=0$, $z_i\ge0$.

\Sec{6. Point-selle, Kuhn--Tucker, dualité}
\B \textbf{Point-selle} $(u,p)\in U\times P$ : $\Lg(u,q)\le\Lg(u,p)\le\Lg(v,p)\ \forall q,v$.
\B \Th{Prop 2.6.2} point-selle de $\Lg=J+q\cdot F$ ($P=\R^M$ ou $\R^M_+$) $\Rightarrow u\in K$ \textbf{min global} (sans convexité !) et $J'(u)+p\cdot F'(u)=0$.
\B \St\Th{Th 2.6.4 Kuhn--Tucker} $J,F_i$ \textbf{convexes} $C^1$, qualif. en $u$ : $u$ min global $\Leftrightarrow\exists p\ge0$ point-selle $\Leftrightarrow$ \textbf{KKT} (CNS). Égalités affines $Av=b\to Av-b\le0,\,b-Av\le0$.
\B \St\textbf{Dualité} : $\mathcal J(v)=\sup_{q\in P}\Lg=J(v)$ si $F(v)\le0$, $+\infty$ sinon (= primal) ; $G(q)=\inf_v\Lg(v,q)$ \textbf{concave} (inf de fct. affines). Primal $\inf\mathcal J$, dual $\sup_{q\ge0}G$. \textbf{Faible} : $\sup G\le\inf\mathcal J$ toujours (écart = \emph{saut de dualité}). \Th{Th 2.6.9} $(u,p)$ selle $\Leftrightarrow\mathcal J(u)=\min\mathcal J=\max G=G(p)$. Saut $>0$ : $\Lg=(v-q)^2$ sur $[0,1]^2$ : $\inf\sup=\frac14>\sup\inf=0$. $U,P$ convexes compacts, $\Lg$ convexe-concave continue $\Rightarrow\exists$ point-selle.
\Mt\Th{Cor 2.6.12} (convexe, qualifié) : (1) $G(q)=\min_v\Lg(v,q)$ explicite, (2) $p=\argmax_{q\ge0}G$, (3) $u=\argmin_v\Lg(v,p)$. $G'(q)=F(u_q)$.
\B QP : $u_q=A^{-1}(b-B\T q)$, $G(q)=-\frac12q\cdot BA^{-1}B\T q+(BA^{-1}b-c)\cdot q-\frac12A^{-1}b\cdot b$ ; $p$ unique si rg $B=M$.

\Sec{7. Algorithmes sans contrainte}
\B \St\textbf{Pas fixe} $u_{n+1}=u_n-\mu J'(u_n)$. \Th{Th 3.2.6} $J$ $\alpha$-conv., $J'$ $L$-Lip, $0<\mu<\frac{2\alpha}{L^2}$ : $\nr{u_n-u}\le\gamma^n\nr{u_0-u}$, $\gamma=\sqrt{1-2\alpha\mu+\mu^2L^2}$ (meilleur $\mu=\alpha/L^2$, $\gamma=\sqrt{1-\alpha^2/L^2}$). \emph{Preuve} : $v_n=u_n-u$, $J'(u)=0$, $\nr{v_{n+1}}^2=\nr{v_n}^2-2\mu\sca{J'(u_n)-J'(u)}{v_n}+\mu^2\nr{J'(u_n)-J'(u)}^2\le(1-2\alpha\mu+L^2\mu^2)\nr{v_n}^2$ (contraction).
\B \Th{Prop 3.3.1} ($C^2$, $\R^N$) CV si $0<\mu<\frac2L$ ; $\mu_{opt}=\frac2{L+\alpha}$, $\gamma=\frac{L-\alpha}{L+\alpha}=\frac{\kappa-1}{\kappa+1}$. Quadratique : CV $\Leftrightarrow0<\mu<\frac2{\lambda_N}$, $\mu_{opt}=\frac2{\lambda_1+\lambda_N}$ ($e_{n+1}=(I-\mu A)e_n$, $\rho<1$). $J(u_n)-J(u)\le C\gamma^{2n}$.
\B Strict. conv., coerc., $J'$ $L$-Lip : CV si $0<\mu<\frac2L$, $J(u_n)-J(u)\le\frac{\nr{u_0-u}^2}{\mu(1-L\mu/2)(n+1)}$ : $O(\frac1n)$. Descente (sans convexité) : $J(v)-J(v-\mu J'(v))\ge\mu(1-\frac{L\mu}2)\nr{J'(v)}^2$. Non convexe : peut CV vers un col ($\frac12x_1^2+\frac14x_2^4-\frac12x_2^2$, $x_0=(1,0)\to O$).
\B \textbf{Pas optimal} $\mu_n=\argmin_\mu J(u_n-\mu J'(u_n))$ : $\sca{J'(u_{n+1})}{J'(u_n)}=0$ (zig-zag). CV si $\alpha$-conv., $J'$ Lip. sur bornés. Quadr. : $\mu_n=\frac{\nr{r_n}^2}{Ar_n\cdot r_n}\in[\frac1{\lambda_N},\frac1{\lambda_1}]$. Armijo : $-p^-j'(0)\mu\le j(0)-j(\mu)\le-p^+j'(0)\mu$. Précond. $P>0$ : $-PJ'$ descente.
\B \textbf{Nesterov} ($J$ convexe, $0<\mu\le\frac1L$) : $u_n=v_n-\mu J'(v_n)$, $a_{n+1}=\frac{1+\sqrt{4a_n^2+1}}2$, $a_0=1$, $v_{n+1}=u_n+\frac{a_n-1}{a_{n+1}}(u_n-u_{n-1})$ : $J(u_n)-J(u)\le\frac{2\mu^{-1}\nr{u_0-u}^2+4(J(u_0)-J(u))}{(n+2)^2}$ : $O(\frac1{n^2})$ \textbf{optimal}. $\alpha$-conv. : $O((1-\sqrt{\alpha/L})^n)$.
\B \textbf{Boule pesante} $u_{n+1}=u_n-\mu J'(u_n)+\nu(u_n-u_{n-1})$ ($m\ddot u+\dot u=-J'$) : locale, $0\le\nu<1$, $0<\mu<\frac{2(1+\nu)}L$ ; $\mu=\frac4{(\sqrt L+\sqrt\alpha)^2}$, $\nu=\gamma^2$, $\gamma=\frac{\sqrt L-\sqrt\alpha}{\sqrt L+\sqrt\alpha}=\frac{\sqrt\kappa-1}{\sqrt\kappa+1}$.
\B \St\textbf{Gradient conjugué} ($A>0$) : $r_0=p_0=Ax_0-b$ ; $\mu_n=\frac{\nr{r_n}^2}{Ap_n\cdot p_n}$, $x_{n+1}=x_n-\mu_np_n$, $r_{n+1}=r_n-\mu_nAp_n$, $\beta_n=\frac{\nr{r_{n+1}}^2}{\nr{r_n}^2}$, $p_{n+1}=r_{n+1}+\beta_np_n$. $Ap_n\cdot p_j=0$ ($j<n$, \textbf{$A$-conjuguées}), $r_n=Ax_n-b$ ; \textbf{exact en $\le N$ itérations} ; $\nr{x_n-x}\le2\sqrt\kappa\big(\frac{\sqrt\kappa-1}{\sqrt\kappa+1}\big)^n\nr{x_0-x}$, $\kappa=\frac{\lambda_N}{\lambda_1}$, sans paramètre. Non quadr. : $u_{n+1}=u_n-\mu_np_n$, $p_{n+1}=J'(u_{n+1})+\beta_np_n$, FR $\beta=\frac{\nr{J'_{n+1}}^2}{\nr{J'_n}^2}$, PR $\beta=\frac{\sca{J'_{n+1}}{J'_{n+1}-J'_n}}{\nr{J'_n}^2}$.
\B \St\textbf{Sous-gradient} $\partial J(x)=\{p:J(y)\ge J(x)+p\cdot(y-x)\,\forall y\}$ convexe fermé ; $=\{J'(x)\}$ si dérivable ; \textbf{$0\in\partial J(x)\Leftrightarrow x$ min} ; $J=\max_\lambda J_\lambda$ : $J_\lambda'(x)\in\partial J(x)$ si $\lambda$ actif ; $\partial|\cdot|(0)=[-1,1]$. Algo $x_{k+1}=x_k-\rho_k\frac{p_k}{\nr{p_k}}$, $\rho_k\to0$, \textbf{$\sum\rho_k=\infty$, $\sum\rho_k^2<\infty$} (ex. $\rho_k=(k+1)^{-1/2-\eps}$). CV ($J$ conv., coerc., loc. Lip.), lent : $\min_{k\le i}J(x_k)-J^*\le Ci^{-1/2+\eps}$. Pas de descente.
\B \textbf{Stochastique} $F=\frac1n\sum f_i$ : $x_{k+1}=x_k-\mu_kf_{i_k}'(x_k)$, $i_k$ uniforme ($\mathbb E f'_{i_k}=F'$), $\mu_k\sim\frac1{k+1}$ ; CV en moyenne si $F$ $\alpha$-conv. et $\frac1n\sum\nr{f_i'}^2\le C(1+\nr{x-x^*}^2)$. Mini-batch.
\B \textbf{Proximal} $\mathrm{prox}_{\tau J}(x)=\argmin_y\frac1{2\tau}\nr{y-x}^2+J(y)$ (gradient \emph{implicite} $x_{k+1}=x_k-\tau J'(x_{k+1})$) : CV $\forall\tau>0$ ; $\alpha$-conv. : $\nr{x_k-x^*}^2\le(1+2\tau\alpha)^{-k}\nr{x_0-x^*}^2$. $\mathrm{prox}_{\tau\nr\cdot_1}=S_\tau$, $S_\tau(b)_i=\mathrm{sgn}(b_i)\max(|b_i|-\tau,0)$ ; $\mathrm{prox}_{I_K}=P_K$ ; $J=\frac12Ax\cdot x-b\cdot x$ : $\mathrm{prox}_{\tau J}(x)=(I+\tau A)^{-1}(x+\tau b)$. Moreau--Yosida $J_\tau(x)=\min_y\frac1{2\tau}\nr{x-y}^2+J(y)$ : convexe, dérivable, même min ; prox = pas de gradient sur $J_\tau$. \textbf{Explicite-implicite} $\min J_1+J_2$ : $x_{k+1}=\mathrm{prox}_{\tau J_2}(x_k-\tau J_1'(x_k))$ ; LASSO $\frac12\nr{Ax-b}^2+\lambda\nr x_1$ : $x_{k+1}=S_{\tau\lambda}(x_k-\tau A\T(Ax_k-b))$ ; $J_2=I_K$ : gradient projeté.
\B \St\textbf{Newton} $F(u)=0$ : $u_{n+1}=u_n-F'(u_n)^{-1}F(u_n)$ (résoudre un système). Min : $u_{n+1}=u_n-J''(u_n)^{-1}J'(u_n)$ = min du modèle quadratique. \Th{Prop 3.5.1} zéro \textbf{régulier} ($F'(u)$ inversible), $\nr{u_0-u}\le\eps$ : $\nr{u_{n+1}-u}\le C\nr{u_n-u}^2$ (\textbf{quadratique, locale}), $C\approx\frac12\nr{F'^{-1}}\nr{F''}$. \pg{Défauts} : coût, locale, peut aller vers max/col ; $F=\nr x^2$ : seulement géométrique. Quasi-Newton : $S_{n+1}=S_n+$rang 1. Gauss--Newton $\min\nr{F}^2$ : $u_{n+1}=u_n-(\nabla F\T\nabla F)^{-1}\nabla F\T F(u_n)$.
\B \textbf{Newton--Lagrange} ($G=0$) : $\begin{psmallmatrix}J''+\lambda G''&G'^{\top}\\G'&0\end{psmallmatrix}\begin{psmallmatrix}\delta u\\\delta\lambda\end{psmallmatrix}=-\begin{psmallmatrix}J'+\lambda G'\\G\end{psmallmatrix}$, inversible si $\mathrm{Hess}\,\Lg>0$ sur $\ker G'$ ($\equiv$ SQP).

\Sec{8. Algorithmes avec contraintes}
\B \St\textbf{Gradient projeté} $u_{n+1}=P_K(u_n-\mu J'(u_n))$, \Th{Th 3.4.1} $0<\mu<\frac{2\alpha}{L^2}$ (contraction $\circ$ $P_K$ 1-Lip). \emph{Faisable} ; il faut $P_K$ explicite.
\B \St\textbf{Uzawa} ($\min J$, $F\le0$) : $p_0\ge0$ ; $u_n=\argmin_v\Lg(v,p_n)$ ; $p_{n+1}=P_{\R^M_+}(p_n+\mu F(u_n))=\max(0,\cdot)$. \Th{Th 3.4.3} $J$ $\alpha$-conv., $F$ convexe $L$-Lip, $0<\mu<\frac{2\alpha}{L^2}$ : $u_n\to u$ (\pg{pas forcément $p_n$}). = gradient projeté (montée) sur le dual ($G'(p)=F(u_p)$). Égalités : $p_{n+1}=p_n+\mu F(u_n)$ sans projection. \emph{Preuve} : $r_n=p_n-p$, $\nr{r_{n+1}}^2\le\nr{r_n}^2+(L^2\mu^2-2\mu\alpha)\nr{u_n-u}^2$. \textbf{Arrow--Hurwicz} : un seul pas de gradient en $v$.
\B \St\textbf{Pénalisation ext.} $J_\eps=J+\frac1\eps\sum\max(F_i,0)^2$ : $u_\eps\to u$ ($J$ cont., strict. conv., coerc. ; $F_i$ conv. cont. ; $K\ne\emptyset$). $\frac2\eps\max(F_i(u_\eps),0)\to\lambda_i$. Mal conditionné si $\eps\to0$ $\to$ continuation. Égalités : $\frac1\eps\nr F^2$. \textbf{Barrière} (int.) : $J-\eps\sum\frac1{F_i}$ ou $-\eps\sum\log(-F_i)$, $F<0$.
\B \textbf{Lagrangien augmenté} $J+\lambda\cdot F+\frac\mu2\nr F^2$ : $v_n=\argmin$, $\lambda_{n+1}=\lambda_n+\mu F(v_n)$ ; pas besoin de $\mu\to\infty$ (si Hess $\Lg>0$ sur $\ker F'$).
\B SLP : linéariser $J,F$ (+ région de confiance $\nr{v-v_{n-1}}_\infty\le\delta$). SQP : modèle quadr. avec \textbf{Hess du Lagrangien}.

\Sec{1. Forme standard, géométrie}
\B \St\textbf{Conversions} : $\max c\cdot x=-\min(-c\cdot x)$ ; $a\cdot x\le\beta\to a\cdot x+s=\beta$, $s\ge0$ ; $a\cdot x\ge\beta\to a\cdot x-s=\beta$, $s\ge0$ ; $x_j$ libre $\to x_j=x_j^+-x_j^-$ ; $x_j\ge\ell\to x_j'=x_j-\ell$ ; $x_j\le0\to-x_j$ ; $=\ \Leftrightarrow\ \le$ et $\ge$ ; $b_i<0$ : ligne $\times(-1)$.
\B $X_{ad}=\{Ax=b,x\ge0\}$ : \textbf{polyèdre} convexe (inter. finie de demi-espaces), éventuellement vide / non borné. \emph{Admissible} = vérifie les contraintes ; \emph{optimale} = atteint le min.
\B \St\Th{Lemme 4.2.3} $\exists$ solution optimale $\Leftrightarrow-\infty<\inf<+\infty$ (cône $\{\sum x_iA_i,x\ge0\}$ fermé, Farkas).
\B \textbf{Sommet} (point extrémal) : $x=\theta y+(1-\theta)z$, $\theta\in]0,1[$, $y,z\in X_{ad}\Rightarrow y=z=x$.
\B \textbf{Base} $B$ : $m$ colonnes indépendantes, $A=[B\ N]$, $x=(x_B,x_N)$ ; \textbf{solution basique} $x_N=0$, $x_B=B^{-1}b$ (\emph{admissible} si $\ge0$ ; \emph{dégénérée} si une $(x_B)_i=0$). Variables de base / hors base.
\B \St\Th{L 4.2.5} sommets $=$ solutions basiques admissibles. \St\Th{Prop 4.2.6} s'il existe un optimum, il existe un \textbf{optimum basique} (sommet). $X_{ad}\ne\emptyset\Rightarrow$ au moins un sommet. Au plus $\binom nm$ bases ; cube $[0,1]^{n-m}$ : $2^{n-m}$ sommets.
\B 2D : tracer les droites, sens de $-c$, l'optimum est en un sommet ; ensemble des optima = une face (convexe).

\Sec{2. Algorithme du simplexe}
\B \St\textbf{Coûts réduits} : $c\cdot x-c\cdot\bar x=\tilde c_N\cdot x_N$, $\tilde c_N=c_N-N\T B^{-\top}c_B=c_N-N\T\pi$, $\pi=B^{-\top}c_B$ (multiplicateurs = var. duale).
\B \St\Th{Prop 4.2.8} $\tilde c_N\ge0\Rightarrow\bar x$ \textbf{optimale} (CNS si non dégénérée ; \pg{seulement suffisante si dégénérée}).
\Mt\textbf{Itération} (base admissible $B$) : (1) $x_B=B^{-1}b$, $\pi=B^{-\top}c_B$, $\tilde c_N$ ; (2) $\tilde c_N\ge0$ : STOP ; (3) \textbf{entrante} $i$ : $\tilde c_i<0$ le plus négatif (Dantzig) ; (4) $d=B^{-1}a_i$ ; si $d\le0$ : \textbf{non borné} ($-\infty$) ; (5) \textbf{test du ratio} $\eps^*=\min_{j:d_j>0}\frac{(x_B)_j}{d_j}$ $\to$ \textbf{sortante} $j$ ; (6) $x_B\leftarrow x_B-\eps^*d$, $x_i=\eps^*$, échanger $i\leftrightarrow j$. Coût $-=\eps^*|\tilde c_i|$.
\Mt\textbf{Tableau} : lignes $[B^{-1}A\mid B^{-1}b]$, dernière $[\tilde c\mid-c_B\cdot x_B]$. Colonne pivot : $\tilde c<0$ ; ligne pivot : min $\bar b_j/\bar a_{ji}$ ($\bar a_{ji}>0$) ; pivot de Gauss--Jordan (pivot$\to1$, reste de la colonne $\to0$, y compris ligne coût). Départ facile : $Ax\le b$, $b\ge0$ $\to$ base = écarts ($x=0$, $s=b$).
\B \textbf{Ex. tableau} (4.2.4) $\min2x_1-x_2$, $x_1+x_2+s_1=1$, $-x_1+x_2+s_2=\frac12$ :\\
{\renewcommand{\arraystretch}{1.0}$\begin{array}{c|rrrr|r}
 &x_1&x_2&s_1&s_2&\\\hline
s_1&1&1&1&0&1\\
s_2&-1&\boxed{1}&0&1&\frac12\\\hline
\tilde c&2&-1&0&0&0
\end{array}\ \to\
\begin{array}{c|rrrr|r}
 &x_1&x_2&s_1&s_2&\\\hline
s_1&2&0&1&-1&\frac12\\
x_2&-1&1&0&1&\frac12\\\hline
\tilde c&1&0&0&1&\frac12
\end{array}$}\\
$x_2$ entre ($-1$), ratios $1,\frac12$ $\to s_2$ sort ; $\tilde c\ge0$ : $\bar x=(0,\frac12)$, val. $=-\frac12$ (case en bas à droite $=-z$) ; dual $p_i=-\tilde c_{s_i}=(0,-1)$.
\B \textbf{Dégénérescence} : $\eps^*=0$, coût stationnaire, \textbf{cyclage} possible (règle de Bland : plus petits indices). Fini si toutes les bases visitées sont non dégénérées (L 4.2.10). Pire cas exponentiel, polynomial en pratique.
\B \textbf{Ex. 4.2.5} $\min3x_3-x_4$, $x_1-3x_3+3x_4=6$, $x_2-8x_3+4x_4=4$ : base $(x_1,x_2)=(6,4)$, $\tilde c_{(3,4)}=(3,-1)$ $\to x_4$ entre, $d=(3,4)$, ratios $\frac63,\frac44$ $\to x_2$ sort ; base $(x_1,x_4)=(3,1)$, coût $=-1+x_3+\frac14x_2$ $\Rightarrow\tilde c\ge0$ : $\bar x=(3,0,0,1)$, val. $-1$, $\pi=(0,-\frac14)$. Ex. 4.2.1 $\max x_1+2x_2$ ($-3x_1+2x_2\le2$, $-x_1+2x_2\le4$, $x_1+x_2\le5$) : $(2,3)$, val. 8. Ex. 4.2.4 $\min2x_1-x_2$ ($x_1+x_2\le1$, $x_2-x_1\le\frac12$) : $(0,\frac12)$, val. $-\frac12$.
\B \St\textbf{Phase I} : rendre $b\ge0$ ($\tilde b_i=|b_i|$), $\min\mathbf1\cdot y\st Ax+y=b$, $x,y\ge0$, départ $(x,y)=(0,b)$. Opt $=0\Leftrightarrow X_{ad}\ne\emptyset$ (base admissible pour (P)) ; opt $>0\Rightarrow X_{ad}=\emptyset$.
\B $B^{-1}$ : $B_{k+1}=B_kE_k$, $E_k=I$ sauf col. $j$ $=l=B_k^{-1}a_i$ ; $E_k^{-1}$ : col. $j=(-\frac{l_1}{l_j},\ldots,\frac1{l_j},\ldots,-\frac{l_n}{l_j})$.
\B \textbf{Points intérieurs} : $\min\mu\pi(x)+c\cdot x$, $Ax=b$, $x>0$, $\pi=-\sum\log x_i$ (barrière, strict. convexe) ; optimalité $c-\frac\mu x+A\T p=0$ ($x_is_i=\mu$, trajectoire centrale) ; Newton, $\mu\to0$ ; $x_\mu\to x_0$ si $X^0_{ad}$ borné $\ne\emptyset$, sol. unique.

\Sec{3. Dualité PL}
\B \St$\Lg(x,p)=c\cdot x+p\cdot(b-Ax)$, $G(p)=\min_{x\ge0}\Lg=p\cdot b$ si $A\T p\le c$, $-\infty$ sinon. \textbf{(D) $\max b\cdot p\st A\T p\le c$, $p\in\R^m$ libre.} Dual du dual = primal.
\B \St\textbf{Tableau} (primal \textbf{min} $\leftrightarrow$ dual \textbf{max}) :\\
{\renewcommand{\arraystretch}{1.05}\begin{tabular}{@{}l|l@{}}
contrainte $a_i\cdot x=b_i$ & $p_i$ libre\\
$a_i\cdot x\ge b_i$ & $p_i\ge0$\\
$a_i\cdot x\le b_i$ & $p_i\le0$\\\hline
variable $x_j\ge0$ & $(A\T p)_j\le c_j$\\
$x_j$ libre & $(A\T p)_j=c_j$\\
$x_j\le0$ & $(A\T p)_j\ge c_j$\\
\end{tabular}}\\
Symétrique : $\min c\cdot x$, $Ax\ge b$, $x\ge0\ \leftrightarrow\ \max b\cdot p$, $A\T p\le c$, $p\ge0$. $\min c\cdot v$, $v\ge0$, $Av\le b\ \leftrightarrow\ \max-b\cdot q$, $q\ge0$, $A\T q+c\ge0$.
\B \St\textbf{Faible} : $x,p$ admissibles $\Rightarrow c\cdot x\ge b\cdot p$ ($c\cdot x\ge A\T p\cdot x=p\cdot Ax=p\cdot b$). Égalité $\Rightarrow$ \textbf{les deux optimaux} (certificat).
\B \St\Th{Th 4.2.13 (forte)} si l'un a une valeur finie, les deux ont des solutions et $\min c\cdot x=\max b\cdot p$. Primal $-\infty\Rightarrow$ dual vide ; dual $+\infty\Rightarrow$ primal vide ; les deux peuvent être vides.
\B \St\textbf{Écarts complémentaires} : $x\cdot(c-A\T p)=0$ : $x_j>0\Rightarrow(A\T p)_j=c_j$ ; $(A\T p)_j<c_j\Rightarrow x_j=0$. Forme $\ge$ : $p_i>0\Rightarrow a_i\cdot x=b_i$ ; $a_i\cdot x>b_i\Rightarrow p_i=0$. KKT de la PL : $Ax=b$, $x\ge0$, $A\T p\le c$, $x\cdot(c-A\T p)=0$ $\Leftrightarrow$ optimalité.
\B $p_i=\partial(\text{val. opt.})/\partial b_i$ : \textbf{prix marginal} (valable tant que la base optimale ne change pas). À l'optimum du simplexe : $\bar p=B^{-\top}c_B$ ($\tilde c_N\ge0\Leftrightarrow\bar p$ dual-admissible). Tableau final ($\min$, $Ax\le b$ + écarts $s$) : coût réduit de $s_i$ $=-p_i$. Sensibilité : la base reste optimale tant que $B^{-1}(b+\Delta b)\ge0$ ($p$ inchangé) et $\tilde c_N\ge0$. \textbf{Farkas (PL)} : $\{Ax=b,x\ge0\}=\emptyset\Leftrightarrow\exists p$ : $A\T p\le0$, $b\cdot p>0$.
\Mt\textbf{Résoudre via le dual} : dual à 2 variables $\to$ dessin $\to$ $\bar p$ ; puis écarts compl. (contrainte duale non serrée $\Rightarrow x_j=0$ ; $\bar p_i\neq0\Rightarrow$ contrainte $i$ serrée) $\to$ système $\to\bar x$ ; vérifier $c\cdot\bar x=b\cdot\bar p$.
\B Ex. 4.2.7 : $\min8x_1+9x_2+4x_3+6x_4$, $4x_1+x_2+x_3+2x_4\ge1$, $x_1+3x_2+2x_3+x_4\ge1$ ; dual $\max p_1+p_2$, $4p_1+p_2\le8$, $p_1+3p_2\le9$, $p_1+2p_2\le4$, $2p_1+p_2\le6$ : $\bar p=(\frac{12}7,\frac87)$, $\bar x=(\frac17,0,\frac37,0)$, val. $\frac{20}7$.
\B Centre d'appels : $x_i$ = opérateurs tranche $i$, $\min S\cdot x\st Ux\ge c$, $x\ge0$ ($U$ d'intervalles, TU). $S=(4,5,6)$, $c=(5,10,6,12,4)$ : $\bar x=(5,8,4)$, $\bar p=(4,0,0,5,1)$, coût $84$.

\Sec{4. Nombres entiers, unimodularité}
\B PLNE $\min c\cdot x$, $x\in P\cap\Z^n$. \textbf{Relaxation} ($x\in P$) : $v_{\text{relax}}\le v_{\text{PLNE}}$ (min) ; arrondir peut être non admissible ou loin de l'optimum.
\B $P$ = enveloppe convexe de ses sommets ; $J$ \textbf{linéaire} (ou concave) : $\min_PJ=\min_{\text{sommets}}J$. \pg{Slide : « $J$ convexe » -- faux pour le min ($\nr x^2$ sur $[-1,1]$) ; vrai pour $\max$.}
\B \St\textbf{Unimodulaire} : $A\in\Z^{n\times n}$ inversible, $\det A=\pm1\Leftrightarrow\forall b\in\Z^n$, $A^{-1}b\in\Z^n$ (Cramer). \textbf{Totalement unimodulaire (TU)} : toute sous-matrice carrée a $\det\in\{0,\pm1\}$ $\Rightarrow$ coeffs $\in\{0,\pm1\}$.
\B \St\Th{Prop 4.3.3} $A$ TU, $b\in\Z^m$ ($c$ quelconque) $\Rightarrow$ toute \textbf{solution basique est entière} $\Rightarrow$ le simplexe donne un optimum entier ; $\min_{P\cap\Z^n}=\min_P$. \pg{Pas « toute solution optimale entière »} (barycentres).
\B \St\Th{L 4.3.5 Poincaré} coeffs $0,\pm1$, \textbf{au plus un $+1$ et au plus un $-1$ par colonne} $\Rightarrow$ TU (récurrence : colonne nulle $\to0$ ; colonne à 1 coeff. $\to$ développer ; sinon somme des lignes $=0\to\det=0$).
\B TU aussi : matrice d'\textbf{intervalles} (1 consécutifs par colonne ou ligne) ; $A\T$, $-A$, $[A\ I]$, $\begin{psmallmatrix}A\\I\end{psmallmatrix}$, permutations, dupliquer lignes. Incidence d'un graphe \textbf{biparti} non orienté. Pas TU : $\begin{psmallmatrix}1&1&0\\0&1&1\\1&0&1\end{psmallmatrix}$ ($\det=2$, cycle impair).

\Sec{5. Flots}
\B Graphe orienté $G=(\mathcal N,\mathcal A)$, $m$ nœuds, $n$ arcs (arbre : connexe sans cycle, $m-1$ arcs) ; capacité $u_{ij}\ge0$, coût $c_{ij}$, flot externe $b_i$ ($>0$ entrant, $<0$ sortant).
\B \St\textbf{Kirchhoff} : $b_i+\sum_{(j,i)\in\mathcal A}x_{ji}=\sum_{(i,j)\in\mathcal A}x_{ij}$ ; nécessaire $\sum_ib_i=0$. \textbf{Matrice d'incidence} $A_{i,l}=+1$ si l'arc $l$ part de $i$, $-1$ s'il arrive en $i$, $0$ sinon : $Ax=b$ ; un $+1$ et un $-1$ par colonne $\Rightarrow$ \textbf{TU} (L 4.3.9) ; rang $m-1$ (graphe connexe) : une ligne redondante.
\B \St\textbf{Flot de coût min} $\min c\cdot x\st Ax=b$, $0\le x\le u$ ; existe si admissible et ($c\ge0$ ou $u<\infty$). Forme std : $z=(x,y)$, $\begin{psmallmatrix}A&0\\I&I\end{psmallmatrix}z=\begin{psmallmatrix}b\\u\end{psmallmatrix}$. \Th{Th 4.3.7} $b,u$ \textbf{entiers} $\Rightarrow$ basiques entières $\Rightarrow$ simplexe $\to$ flot optimal \textbf{entier}. Bases = arbres couvrants.
\B \textbf{Transport} : entrepôts $i$ ($b_i=s_i$) $\to$ clients $j$ ($b_j=-r_j$), arcs $i\to j$ coût $c_{ij}$, $u=\infty$ ; si $\sum s>\sum r$ : \textbf{client fictif} coût 0. Ex. usines $(100,80)$, demandes $(80,75)$, $c=\begin{psmallmatrix}8&15\\10&4\end{psmallmatrix}$ : $x_{11}=80$, $x_{22}=75$, fictif $(20,5)$, coût $940$. \textbf{Simplexe du transport} : init. \emph{coin nord-ouest} ($x_{11}=\min(s_1,r_1)$, puis avancer ligne/colonne) ; $M+N-1$ cases de base (arbre) ; potentiels $u_i+v_j=c_{ij}$ sur la base, $\tilde c_{ij}=c_{ij}-u_i-v_j$ ; faire entrer une case $\tilde c_{ij}<0$, cycle $+\theta/-\theta$, $\theta$ = min des cases $-\theta$.
\B \textbf{Flot max} : source $s$ (sans prédécesseur), puits $p$ (sans successeur), $b_s=v=-b_p$, $\max v$ ; ajouter l'arc $(p,s)$, $b=0$, coûts $0$ sauf $c_{ps}=-1$ ($\max x_{ps}$) ; les $c_{ij}$ n'interviennent pas. (Hors slides) \textbf{Ford--Fulkerson} : chemin augmentant $s\to p$ dans le graphe résiduel (arcs avant non saturés, arcs arrière de flot $>0$), augmenter du min des résidus ; \textbf{flot max = coupe min} $\sum_{i\in S,j\notin S}u_{ij}$ ($s\in S$, $p\notin S$).
\B \textbf{Plus court chemin} : $b_s=1=-b_p$, $c_{ij}\ge0$, $u=\infty$ ; TU $\Rightarrow x_{ij}\in\{0,1\}$ ; solutions connexes. Dual : distances $d$, $\max d_p-d_s\st d_j-d_i\le c_{ij}$. \textbf{Dijkstra} (plus efficace, $c\ge0$) : $d_s=0$, $d_i=\infty$ ; répéter : marquer le nœud non marqué $i$ de $d_i$ min, puis $\forall(i,j)$ : $d_j\leftarrow\min(d_j,d_i+c_{ij})$ ; $O(m^2)$.
\B \textbf{Affectation} $\max\sum a_{ij}v_{ij}\st\sum_jv_{ij}=1$, $\sum_iv_{ij}=1$, $v\ge0$ (bistochastiques) : TU $\Rightarrow$ optimum = permutation ($N!$ possibilités ; Birkhoff--von Neumann).

\Sec{6. Problèmes entiers célèbres}
\B \St\textbf{Sac-à-dos} $\max u\cdot x\st p\cdot x\le L$, $x\in\{0,1\}^N$ (NP-complet). \textbf{Glouton} : trier $u_j/p_j\downarrow$, prendre si ça tient $\to$ borne \emph{inf.} (peut être très mauvais : $\max$(glouton, meilleur objet seul) $\ge\frac12$opt). \textbf{Relaxation continue (Dantzig)} : même ordre, objet critique $k$ fractionnaire $x_k=\frac{L-\sum_{j<k}p_j}{p_k}$ $\to$ borne \emph{sup.} ($\lambda=u_k/p_k$ = multiplicateur). \emph{Ex.} $L=14$, $(u,p)$ : A$(16,4)$ B$(15,4)$ C$(22,8)$ D$(4,2)$ E$(12,8)$ : glouton $\{A,B,D\}=35$ ; Dantzig $x=(1,1,\frac34,0,0)\to47{,}5$ ; donc $35\le$ opt $\le47$ ; B\&B (séparer sur C) : opt $\{A,C,D\}=42$ en 7 nœuds.
\B \St\textbf{Séparation--évaluation (B\&B)} : arbre ($x_k=0$ / $x_k=1$) ; évaluer chaque nœud par une \textbf{relaxation} (continue ou lagrangienne) ; \textbf{élaguer} si borne $\le$ meilleure connue (max) / $\ge m$ (min) ; relaxation entière $\to$ candidate ; irréalisable $\to$ fermer ; sinon séparer sur une variable fractionnaire. Initialiser avec une heuristique (glouton). « + coupe » : ajouter des inégalités valides qui coupent la solution fractionnaire ; \textbf{Gomory} : ligne finale $x_i+\sum_j\bar a_{ij}x_j=\bar b_i$, $\bar b_i\notin\Z$ $\Rightarrow\sum_j\{\bar a_{ij}\}x_j\ge\{\bar b_i\}$ ($\{\cdot\}$ = partie fractionnaire).
\Pg En \textbf{min} (TSP) l'heuristique donne une borne \textbf{SUP} $m$ (la slide dit « inférieure »), la relaxation une borne INF. La relaxation est optimiste, la solution connue réaliste.
\B \textbf{Relaxation lagrangienne} : $\theta(\lambda)=\max_{x\in X}u\cdot x-\lambda g(x)\ge v$ ($\lambda\ge0$) ; $\min_\lambda\theta$ (convexe, sous-gradient $-g(x_\lambda)$). Sac-à-dos : $\theta(\lambda)=\lambda L+\sum_j\max(0,u_j-\lambda p_j)$, min en $\lambda=u_k/p_k$ = borne de Dantzig. TSP 4 villes ($d_{12}=4,d_{13}=7,d_{14}=5,d_{23}=3,d_{24}=6,d_{34}=2$) : plus proche voisin $m=14$, borne racine 11, nœud $1\to3$ borne $18\ge14$ élagué ; opt 14.
\B \St\textbf{TSP} : $\min\sum d_{ij}x_{ij}$, $x\in\{0,1\}^{\mathcal A}$, $\sum_jx_{ij}=2\ \forall i$ (degré), $\sum_{(i,j)\in S}x_{ij}\le|S|-1\ \forall S\subsetneq\mathcal N$ (pas de sous-tour ; $2^N$ contraintes). $(N-1)!/2$ tournées. Sans sous-tours : cycles disjoints.
\B Heuristiques : glouton, plus proche voisin, 2-opt, \textbf{recuit simulé} (accepter $\Delta>0$ avec proba $e^{-\Delta/T}$, $T\searrow$).

\Sec{9. Compléments continus : exemples types}
\B \textbf{Pas fixe, quadratique} $J=\frac12(x_1^2+10x_2^2)$ : $\alpha=1$, $L=10$ ; CV $\Leftrightarrow0<\mu<\frac2{10}$ ; $\mu_{opt}=\frac2{11}$, $\gamma=\frac9{11}$ ; GC : 2 itérations.
\B \textbf{KKT (signe)} $\min x_1^2+x_2^2\st x_1+x_2\ge2$ : $F=2-x_1-x_2\le0$ (affine $\Rightarrow$ qualifiée) ; $2x-\lambda(1,1)=0$ ; $\lambda=0\Rightarrow x=0$ non admissible ; active : $x=(1,1)$, $\lambda=2\ge0$ ; convexe $\Rightarrow$ min global, $J=2$. Sensibilité : $x_1+x_2\ge c$ donne $J_{\min}=\frac{c^2}2$, $J_{\min}'(2)=2=\lambda$.
\B \textbf{Dual + Uzawa} $\min\frac12\nr{x-x_0}^2\st Ax\le b$ : $\Lg=\frac12\nr{x-x_0}^2+q\cdot(Ax-b)$, $x_q=x_0-A\T q$, $G(q)=-\frac12\nr{A\T q}^2+q\cdot(Ax_0-b)$ ; Uzawa : $x_n=x_0-A\T q_n$, $q_{n+1}=\max(0,q_n+\mu(Ax_n-b))$, $0<\mu<\frac2{\nr A^2}$ ($\alpha=1$, $L=\nr A$).
\Sec{10. Méthodo examen \& pièges}
\Mt\textbf{Existence} : fermé + continue + coercive (ou compact) ; \textbf{unicité} : $K$ convexe + $J$ strict. convexe ; \textbf{caractériser} : Euler / Lagrange / KKT (vérifier la \textbf{qualification}) ; \textbf{suffisance} : convexité (ou ordre 2 sur le Lagrangien) ; \textbf{calculer} ; \textbf{algorithme} + hypothèses ($\alpha$, $L$, pas).
\Mt\textbf{Dual d'un pb convexe} : $\Lg=J+q\cdot F$ ; $\min_v$ (souvent explicite) $\to G(q)$ ; contraintes duales : $q\ge0$ (inég.), libre (égal.) + domaine où $G>-\infty$.
\Pg KKT = CN (qualif.) ; CNS si convexe. $\lambda\ge0$ seulement pour inégalités $F\le0$ d'un \textbf{min}. Multiplicateurs d'égalité libres. Ordre 2 : Hess $\Lg$ sur $\ker F'$, pas $J''$.
\Pg Simplexe : vérifier l'admissibilité de départ ; ratio seulement sur $d_j>0$ ; $d\le0$ = non borné. Dual d'un \textbf{max} : utiliser le tableau dans l'autre sens (ou passer en min).
